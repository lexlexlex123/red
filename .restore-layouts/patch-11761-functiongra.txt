        function grassSwayChunk(markup,seed){
          const painted='<g fill="'+fill+'">'+paintForestMarkup(markup,fill)+'</g>';
          const innerTf='translate('+f(-pivotX)+','+f(-pivotY)+')'+(gxf?' '+gxf:'');
          if(!doAnimate){
            return '<g transform="'+innerTf+'">'+painted+'</g>';
          }
          // Один простой цикл skew — без лишних ключевых кадров
          const amp=(0.55+rng(seed)*0.7).toFixed(2);
          const dur=(4.5+rng(seed+3)*3.5).toFixed(2);
          const beg=(-rng(seed+5)*parseFloat(dur)).toFixed(2);
          return '<g transform="translate('+f(pivotX)+','+f(pivotY)+')">'+
            '<g>'+
            '<animateTransform attributeName="transform" type="skewX" values="0;'+amp+';'+(-amp)+';0" keyTimes="0;0.35;0.7;1" dur="'+dur+'s" begin="'+beg+'s" repeatCount="indefinite" calcMode="spline" keySplines="0.42 0 0.58 1;0.42 0 0.58 1;0.42 0 0.58 1"/>'+
            '<g transform="'+innerTf+'">'+painted+'</g>'+
            '</g></g>';
        }
