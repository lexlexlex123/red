      function fogFlameBand(yFloor,yCeil,seed,dur,layerOp,extraY){
        extraY=extraY||0;
        const yF=yFloor+fogYOff+extraY, yC=yCeil+fogYOff+extraY;
        const d0=fogFlamePath(seed,yF,yC);
        const op=(layerOp!=null?layerOp:fogOp).toFixed(2);
        let el='<path d="'+d0+'" fill="url(#'+uid+'fogG)" filter="url(#'+uid+'fogBlur)" opacity="'+op+'"';
        if(doAnimate){
          const dkf=fogFlameKF(seed,yF,yC,12);
          el+='><animate attributeName="d" dur="'+dur+'s" repeatCount="indefinite" calcMode="linear" values="'+dkf+'"/></path>';
        } else el+='/>';
        return el;
      }

      // Главный: дымка перед слоем «19» (canopy)
      function mistBeforeLayer2(extraY){
        extraY=extraY||0;
        return fogFlameBand(h*1.08,h*0.20,0,5.5,fogOp,extraY)+
          fogFlameBand(h*1.02,h*0.38,1.9,6.2,fogOp,extraY);
      }

      function mistBeltMid(){
        return fogFlameBand(h*1.05,h*0.16,2.4,5.0,fogOp)+
          fogFlameBand(h*1.02,h*0.30,4.1,5.8,fogOp);
      }

      function mistBeltFront(){
        return fogFlameBand(h*1.18,h*0.40,0.7,4.6,fogOp)+
          fogFlameBand(h*1.15,h*0.56,3.2,5.2,fogOp)+
          fogFlameBand(h*1.10,h*0.72,5.0,6.0,fogOp);
      }

      // Побочный: две дымки — после травы и передний низ
      function mistContentAfterGrass(){
        return fogFlameBand(h*1.10,h*0.72,5.0,6.0,fogOp);
      }

      function mistContentFrontBottom(){
        return fogFlameBand(h*1.18,h*0.40,0.7,4.6,fogOp);
      }
