      const fogBlurStd=Math.max(3, 38*(h/675));
      const mistExtraY=h*90/675;

      // Органическая волна (алгоритм flamePath из темы «Огонь»).
      function fogFlamePath(seed,yFloor,yCeil){
        const pts=14;
        const overshoot=w*0.18;
        const step=(w+overshoot*2)/pts;
        const bandH=Math.max(8,yFloor-yCeil);
        const segs=[];
        for(let i=0;i<=pts;i++){
          const phase=(i/pts)*Math.PI*2+seed;
          const sway=Math.sin(phase)*w*0.09;
          const yVar=Math.sin(phase*2.1+seed*0.8)*bandH*0.12;
          const edge=1-Math.pow((i/pts-0.5)*2,2)*0.75;
          const y=Math.max(yCeil+bandH*0.08,yCeil+yVar+(1-edge)*bandH*0.35);
          segs.push({x:-overshoot+i*step+sway,y});
        }
        let d='M'+f(-overshoot)+','+f(yFloor)+' ';
        segs.forEach((p,i)=>{
          if(i===0) d+='L'+f(p.x)+','+f(p.y)+' ';
          else{
            const prev=segs[i-1];
            d+='Q'+f((prev.x+p.x)/2)+','+f((prev.y+p.y)/2)+' '+f(p.x)+','+f(p.y)+' ';
          }
        });
        return d+'L'+f(w+overshoot)+','+f(yFloor)+' L'+f(-overshoot)+','+f(yFloor)+' Z';
      }

      function fogFlameKF(seed,yFloor,yCeil,n){
        const frames=Array.from({length:n},(_,i)=>fogFlamePath(seed+i*Math.PI*2/n,yFloor,yCeil));
        frames.push(frames[0]);
        return frames.join(';');
      }

      function fogFlameBand(yFloor,yCeil,seed,dur,layerOp,extraY){
        extraY=extraY||0;
        const yOff=fogYOff+fogYContent+extraY;
        const yF=yFloor+yOff, yC=yCeil+yOff;
        const d0=fogFlamePath(seed,yF,yC);
        const op=(layerOp!=null?layerOp:fogOp).toFixed(2);
        let el='<path d="'+d0+'" fill="url(#'+uid+'fogG)" filter="url(#'+uid+'fogBlur)" opacity="'+op+'"';
        if(doAnimate){
          const dkf=fogFlameKF(seed,yF,yC,12);
          const pathDur=(dur*2).toFixed(1);
          const opDur=((5.5+rng(seed+7)*3.5)*2).toFixed(1);
          const opBeg=(-rng(seed+11)*opDur).toFixed(2);
          el+='><animate attributeName="d" dur="'+pathDur+'s" repeatCount="indefinite" calcMode="linear" values="'+dkf+'"/>'+
            '<animate attributeName="opacity" values="0.8;0.5;0.7;0.8;0.2;0.4;0.8" keyTimes="0;0.167;0.333;0.5;0.667;0.833;1" dur="'+opDur+'s" begin="'+opBeg+'s" repeatCount="indefinite" calcMode="spline" keySplines="0.45 0 0.55 1;0.45 0 0.55 1;0.45 0 0.55 1;0.45 0 0.55 1;0.45 0 0.55 1;0.45 0 0.55 1"/></path>';
        } else el+='/>';
        return el;
      }
